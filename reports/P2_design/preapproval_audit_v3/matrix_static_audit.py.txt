def reject_duplicate(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError('Duplicate JSON key: ' + key)
        result[key] = value
    return result


def read(path):
    raw = (ROOT / path).read_bytes()
    text = raw.decode('utf-8', errors='strict')
    sources.append(dict(path=path, sha256=hashlib.sha256(raw).hexdigest(),
                        size_bytes=len(raw), full_bytes_read=True,
                        line_count=len(text.splitlines()),
                        scope='Existing source/design/seed METADATA only; no realizations'))
    if path.endswith('.json'):
        return json.loads(text, object_pairs_hook=reject_duplicate,
                          parse_constant=lambda s: (_ for _ in ()).throw(ValueError(s)))
    return text


def canonical_bytes(value):
    # Audit-only equality digest of parsed design metadata, not production identity.
    return (json.dumps(value, ensure_ascii=True, sort_keys=True,
                       separators=(',', ':'), allow_nan=False) + '\n').encode()


def audit_digest(value):
    return hashlib.sha256(canonical_bytes(value)).hexdigest()


def differences(actual, expected, pointer=''):
    """Exact type/value/key comparison: bool never silently equals integer 1."""
    if type(actual) is not type(expected):
        return [dict(pointer=pointer, reason='TYPE_MISMATCH',
                     actual_type=type(actual).__name__, expected_type=type(expected).__name__)]
    if isinstance(expected, dict):
        errors = []
        for key in sorted(set(actual) - set(expected)):
            errors.append(dict(pointer=pointer + '/' + key, reason='UNEXPECTED_KEY'))
        for key in sorted(set(expected) - set(actual)):
            errors.append(dict(pointer=pointer + '/' + key, reason='MISSING_KEY'))
        for key in sorted(set(expected) & set(actual)):
            errors += differences(actual[key], expected[key], pointer + '/' + key)
        return errors
    if isinstance(expected, list):
        errors = [] if len(actual) == len(expected) else [dict(pointer=pointer, reason='LENGTH_MISMATCH')]
        for index, (a, e) in enumerate(zip(actual, expected)):
            errors += differences(a, e, pointer + '/' + str(index))
        return errors
    return [] if actual == expected else [dict(pointer=pointer, reason='VALUE_MISMATCH',
                                             actual=actual, expected=expected)]

    (OUT / name).write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
