18: def simulate(workload: Workload, method: str,
19:              policy: Callable[[Observation], Proposal | None] | None = None) -> dict:
20:     """Execute only the supplied input; never derive a seed or materialize a job.
21: 
22:     Optional policies are for synthetic negative tests. Normal B1–B5 accept only
23:     public observations. This function exposes no queue or realized duration to
24:     the policy, and makes no performance/host-timing claims.
25:     """
26:     if method not in {"B1", "B2", "B3", "B4", "B5"}:
27:         raise ValueError("unknown baseline")
28:     queue = [(s.arrival, 1, s.id, "", "") for s in workload.services]
29:     heapq.heapify(queue)
30:     services = {s.id: s for s in workload.services}
31:     nodes = {(s.id, n.id): n for s in workload.services for n in s.nodes}
32:     running: dict[tuple[str, str], dict] = {}
33:     completed: dict[tuple[str, str], Completed] = {}
34:     events, executions, decisions = [], [], []
35:     now = 0.0
36:     controller = policy if policy is not None else lambda view: choose(view, method)
37:     while queue or running or len(completed) < len(nodes):
38:         # Full batches are drained before the first observation at this time.
39:         while queue and queue[0][0] == now:
40:             at, phase, service, node, resource = heapq.heappop(queue)
41:             if phase == 0:
42:                 row = running.pop((service, node))
43:                 completed[service, node] = Completed(service, node, resource, at)
44:                 events.append(dict(time=at, kind="complete", service=service, node=node, resource=resource))
45:             else:
46:                 events.append(dict(time=at, kind="arrival", service=service, node="", resource=""))
47:         while True:
48:             view = observe(workload, now,
49:                            tuple(Running(s, n, r["resource"], r["start"], now-r["start"])
50:                                  for (s, n), r in running.items()), tuple(completed.values()))
51:             proposal = controller(view)
52:             decisions.append(dict(time=now, observation=asdict(view),
53:                                   proposal=asdict(proposal) if proposal is not None else None))
54:             if proposal is None:
55:                 break
56:             key = proposal.service, proposal.node
57:             if key not in dependency_ready(view):
58:                 raise ModelError("policy attempted non-ready/duplicate task")
59:             node = nodes[key]
60:             cost = node.cost(proposal.resource)
61:             if any(r["resource"] == proposal.resource for r in running.values()):
62:                 raise ModelError("policy attempted occupied resource")
63:             for edge in node.predecessors:
64:                 pred = completed[proposal.service, edge.node]
65:                 if now < pred.finish + communication(workload, pred.resource, proposal.resource, edge.size):
66:                     raise ModelError("policy attempted start before data ready")
67:             finish = now + cost.realized
68:             number(finish, "actual completion", positive=True)
69:             if finish <= now:
70:                 raise ModelError("duration below representable virtual-time resolution")
71:             row = dict(service=proposal.service, node=proposal.node, resource=proposal.resource,
72:                        start=now, finish=finish)
73:             running[key] = row
74:             executions.append(row.copy())
75:             events.append(dict(time=now, kind="start", service=proposal.service,
76:                                node=proposal.node, resource=proposal.resource))
77:             heapq.heappush(queue, (finish, 0, proposal.service, proposal.node, proposal.resource))
78:         if len(completed) == len(nodes) and not queue:
79:             break
80:         # Data arrival is a wakeup, not a fabricated resource reservation.
81:         wakeups = []
82:         busy = {r["resource"] for r in running.values()}
83:         for service, node_id in dependency_ready(view):
84:             node = nodes[service, node_id]
85:             for cost in node.durations:
86:                 if cost.resource in busy:
87:                     continue
88:                 ready = max((completed[service, e.node].finish + communication(
89:                     workload, completed[service, e.node].resource, cost.resource, e.size)
90:                     for e in node.predecessors), default=services[service].arrival)
91:                 if ready > now:
92:                     wakeups.append(ready)
93:         candidates = ([queue[0][0]] if queue else []) + wakeups
94:         if not candidates:
95:             raise DeadlockError("unfinished admitted nodes, no legal start or future event/data wakeup")
96:         next_time = min(candidates)
97:         if next_time <= now:
98:             raise DeadlockError("event progression failed")
99:         now = next_time
100:     return dict(schema_version=1, origin="HAND_AUTHORED_CORRECTNESS_FIXTURE", method=method,
101:                 executions=executions, events=events, decisions=decisions,
102:                 metrics=summarize(workload, executions))
