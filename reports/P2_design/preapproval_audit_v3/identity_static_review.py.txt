def digest(b):
    return hashlib.sha256(b).hexdigest()

def reject_duplicate_pairs(pairs):
    d = {}
    for k, v in pairs:
        if k in d:
            raise ValueError('duplicate JSON key: ' + k)
        d[k] = v
    return d

def topo(nodes, edges):
    outgoing = {n: [] for n in nodes}
    indegree = {n: 0 for n in nodes}
    for a, b in edges:
        outgoing[a].append(b)
        indegree[b] += 1
    q = collections.deque(n for n in nodes if indegree[n] == 0)
    order = []
    while q:
        a = q.popleft()
        order.append(a)
        for b in outgoing[a]:
            indegree[b] -= 1
            if not indegree[b]: q.append(b)
    return order
