# Trace every new JSON delta to an original pointer; no matrix duplication.
def diff(a,b,path=''):
 out=[]
 if type(a)!=type(b):return [dict(pointer=path or '/',before=a,after=b)]
 if isinstance(a,dict):
  for k in sorted(a.keys()|b.keys()):
   q=path+'/'+k.replace('~','~0').replace('/','~1')
   if k not in a:out.append(dict(pointer=q,before_status='ABSENT',after=b[k]))
   elif k not in b:out.append(dict(pointer=q,before=a[k],after_status='ABSENT'))
   else:out+=diff(a[k],b[k],q)
 elif isinstance(a,list):
  if len(a)!=len(b):out.append(dict(pointer=path,before=a,after=b))
  else:
   for i,(u,v) in enumerate(zip(a,b)):out+=diff(u,v,path+'/'+str(i))
 elif a!=b:out.append(dict(pointer=path,before=a,after=b))
 return out
